cmake_minimum_required(VERSION 3.16)
project(qwen2_5_bmrt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NNTOOLCHAIN_ROOT "" CACHE PATH "SOPHON SDK / nntoolchain containing include and lib")
if(NOT NNTOOLCHAIN_ROOT AND DEFINED ENV{TPU_MLIR_ROOT})
    set(NNTOOLCHAIN_ROOT "$ENV{TPU_MLIR_ROOT}/third_party/nntoolchain")
endif()
if(NOT NNTOOLCHAIN_ROOT)
    message(FATAL_ERROR "Set NNTOOLCHAIN_ROOT to a matching SOPHON SDK, or set TPU_MLIR_ROOT")
endif()
find_path(BMRT_INCLUDE bmruntime_interface.h PATHS "${NNTOOLCHAIN_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(BMRT_LIBRARY NAMES bmrt libbmrt.so.1.0 PATHS "${NNTOOLCHAIN_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_library(BMLIB_LIBRARY NAMES bmlib libbmlib.so.0 PATHS "${NNTOOLCHAIN_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_executable(qwen_bmrt worker.cpp)
target_include_directories(qwen_bmrt PRIVATE "${BMRT_INCLUDE}")
target_link_libraries(qwen_bmrt PRIVATE "${BMRT_LIBRARY}" "${BMLIB_LIBRARY}")
target_compile_options(qwen_bmrt PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(qwen_bmrt PROPERTIES BUILD_RPATH "${NNTOOLCHAIN_ROOT}/lib")
